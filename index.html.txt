  /* ── TOGGLE PRIMARY SECURITY ──
   * Only visible/active when the card is in a group.
   * Turning ON: clears any other primary in same group, sets this one.
   * Turning OFF: clears primary for whole group → all spokes disappear (links remain but undirected).
   * Actually per spec: "If i unselect primary, all spokes are removed and all properties unlink."
   * So toggling off primary = dissolve the group entirely.
   */
  window.lsmTogglePrimary=function(id){
    var p=lsmProps.find(function(x){return x.id===id;});
    if(!p) return;
    var group=lsmGetGroupOf(id);
    if(!group) return; // not in a group; badge should be hidden

    if(group.primaryId===id){
      // Turning OFF primary → dissolve entire group
      lsmDissolveGroup(group);
      lsmRerenderAll(); lsmUpdateLinks(); lsmUpdateSummary();
      lsmShowHint('Primary removed — group unlinked. Re-link properties to reconnect.');
    } else {
      // Turning ON primary for this card
      group.primaryId=id;
      lsmRerenderAll(); lsmUpdateLinks();
      lsmShowHint('★ '+p.name+' set as Primary Security');
    }
  };

    // If less than 2 remain, dissolve

    lsmRerenderAll(); lsmUpdateLinks(); lsmUpdateSummary();

   * 1. User clicks "Link" → enter link mode
   * 2. User clicks cards to select (toggle)
   * 3. User clicks "Link Selected" → confirm
   * 4. If any selected card is already in a group → merge all into that group
   *    Otherwise → create new group
   * 5. Primary badge appears on all cards in group; user picks one
   */
  window.lsmToggleLinkMode=function(){
    lsmLinkMode=!lsmLinkMode;
    lsmLinkSelected.clear();
    var btn=document.getElementById('lsm-link-mode-btn');
    var panel=document.getElementById('lsm-link-panel');
    var canvas=document.getElementById('lsm-canvas-wrap');

    if(lsmLinkMode){
      btn.classList.add('lsm-active');
      panel.classList.add('lsm-show');
      canvas.classList.add('lsm-link-mode');
      lsmUpdateLinkPanel();
      lsmShowHint('Click each property to include in the cross-security group',4000);
    } else {
      btn.classList.remove('lsm-active');
      panel.classList.remove('lsm-show');
      canvas.classList.remove('lsm-link-mode');
      // Clear selection highlights
      lsmProps.forEach(function(p){
        var el=document.getElementById('lsm-prop-'+p.id);
        if(el) el.classList.remove('lsm-selected');
      });
    }
  };

  function lsmUpdateLinkPanel(){
    var count=lsmLinkSelected.size;
    document.getElementById('lsm-link-count').textContent=count+' selected';
  }

  /* Called when a card is clicked in link mode */
  function lsmHandleLinkClick(id){
    if(!lsmLinkMode) return;
    var el=document.getElementById('lsm-prop-'+id);
    if(!el) return;
    if(lsmLinkSelected.has(id)){
      lsmLinkSelected.delete(id);
      el.classList.remove('lsm-selected');
    } else {
      lsmLinkSelected.add(id);
      el.classList.add('lsm-selected');
    }
    lsmUpdateLinkPanel();
  }

  window.lsmConfirmLinkGroup=function(){
    if(lsmLinkSelected.size<2){
      lsmShowHint('Select at least 2 properties to link');
      return;
    }
    // Collect all selected ids
    var selectedIds=Array.from(lsmLinkSelected);

    // Find any existing groups that contain these ids
    var affectedGroups=[];
    selectedIds.forEach(function(id){
      var g=lsmGetGroupOf(id);
      if(g&&affectedGroups.indexOf(g)===-1) affectedGroups.push(g);
    });

    if(affectedGroups.length===0){
      // New group
      lsmGroups.push({ids:new Set(selectedIds),primaryId:null});
    } else if(affectedGroups.length===1){
      // Merge selected into existing group
      selectedIds.forEach(function(id){ affectedGroups[0].ids.add(id); });
    } else {
      // Merge multiple groups + selected into one
      var masterGroup=affectedGroups[0];
      // Take primaryId from first group that has one
      var existingPrimary=null;
      affectedGroups.forEach(function(g){if(g.primaryId&&!existingPrimary) existingPrimary=g.primaryId;});
      affectedGroups.slice(1).forEach(function(g){
        g.ids.forEach(function(id){ masterGroup.ids.add(id); });
        lsmGroups=lsmGroups.filter(function(x){return x!==g;});
      });
      selectedIds.forEach(function(id){ masterGroup.ids.add(id); });
      if(existingPrimary) masterGroup.primaryId=existingPrimary;
    }

    lsmToggleLinkMode(); // exit link mode, clear highlights
    lsmRerenderAll(); lsmUpdateLinks(); lsmUpdateSummary();
    lsmShowHint('Properties linked ✓ — click ★ Set Primary on the main security card',3500);
  };

  /* ── DRAG ── */
  function lsmMakeDraggable(el,p){
    var sX,sY,sL,sT,dragging=false;
    el.addEventListener('mousedown',function(e){
      if(!e.target.closest('.lsm-card-header')) return;
      if(e.target.closest('.lsm-icon-btn')||e.target.closest('.lsm-prop-type-badge')||e.target.closest('.lsm-primary-badge')) return;
      if(lsmLinkMode) return;
      dragging=true; sX=e.clientX; sY=e.clientY; sL=p.x; sT=p.y;
      el.classList.add('lsm-dragging'); e.preventDefault();
    });
    document.addEventListener('mousemove',function(e){
      if(!dragging) return;
      p.x=Math.max(0,sL+(e.clientX-sX)); p.y=Math.max(0,sT+(e.clientY-sY));
      el.style.left=p.x+'px'; el.style.top=p.y+'px';
      lsmUpdateLinks();
    });
    document.addEventListener('mouseup',function(){
      if(dragging){ dragging=false; el.classList.remove('lsm-dragging'); }
    });
    el.addEventListener('touchstart',function(e){
      if(!e.target.closest('.lsm-card-header')||lsmLinkMode) return;
      dragging=true; sX=e.touches[0].clientX; sY=e.touches[0].clientY; sL=p.x; sT=p.y;
      el.classList.add('lsm-dragging');
    },{passive:true});
    document.addEventListener('touchmove',function(e){
      if(!dragging) return;
      p.x=Math.max(0,sL+(e.touches[0].clientX-sX)); p.y=Math.max(0,sT+(e.touches[0].clientY-sY));
      el.style.left=p.x+'px'; el.style.top=p.y+'px';
      lsmUpdateLinks();
    },{passive:true});
    document.addEventListener('touchend',function(){
      if(dragging){ dragging=false; el.classList.remove('lsm-dragging'); }
    });
  }

  /* ── LINK LINES ──
   * Rules:
   * - Only draw lines when a group has a primaryId set
   * - Draw one spoke from each non-primary member to the primary
   * - All spokes are BLUE, all labelled "CROSS SECURED"
   * - Arrow points TO the primary
   * - No primary set → no lines drawn at all
   */
  function lsmUpdateLinks(){
    var svg=document.getElementById('lsm-links-layer');
    svg.innerHTML='';
    var canvas=document.getElementById('lsm-canvas');
    var maxX=lsmProps.reduce(function(m,p){return Math.max(m,p.x+320);},900);
    var maxY=lsmProps.reduce(function(m,p){return Math.max(m,p.y+450);},600);
    canvas.style.minWidth=maxX+'px'; canvas.style.minHeight=maxY+'px';
    var ns='http://www.w3.org/2000/svg';

    lsmGroups.forEach(function(group){
      if(!group.primaryId) return; // no primary = no spokes
      var primaryProp=lsmProps.find(function(p){return p.id===group.primaryId;});
      if(!primaryProp) return;

      var lineColor='#2960cc';
      var glowColor='rgba(41,96,204,0.15)';
      var labelColor='#7fb3ff';
      var labelBorder='#2960cc';
      var labelText='CROSS SECURED';
      var labelWidth=112;

      group.ids.forEach(function(id){
        if(id===group.primaryId) return; // no self-spoke
        var spokeProp=lsmProps.find(function(p){return p.id===id;});
        if(!spokeProp) return;

        // spoke: from non-primary → primary
        var x1=spokeProp.x+144, y1=spokeProp.y+70;
        var x2=primaryProp.x+144, y2=primaryProp.y+70;
        var mx=(x1+x2)/2, my=(y1+y2)/2;

        // Glow
        var glow=document.createElementNS(ns,'path');
        glow.setAttribute('d','M'+x1+' '+y1+' C'+mx+' '+y1+' '+mx+' '+y2+' '+x2+' '+y2);
        glow.setAttribute('stroke',glowColor); glow.setAttribute('stroke-width','8'); glow.setAttribute('fill','none');
        svg.appendChild(glow);

        // Main bezier
        var path=document.createElementNS(ns,'path');
        path.setAttribute('d','M'+x1+' '+y1+' C'+mx+' '+y1+' '+mx+' '+y2+' '+x2+' '+y2);
        path.setAttribute('stroke',lineColor); path.setAttribute('stroke-width','2'); path.setAttribute('fill','none');
        svg.appendChild(path);

        // Arrowhead at primary (x2,y2)
        var dx=x2-mx, dy=y2-my;
        var len=Math.sqrt(dx*dx+dy*dy)||1; dx/=len; dy/=len;
        var arrowSize=9;
        var ax=x2-dx*arrowSize, ay=y2-dy*arrowSize;
        var arrow=document.createElementNS(ns,'polygon');
        arrow.setAttribute('points',x2+','+y2+' '+(ax+dy*arrowSize*0.5)+','+(ay-dx*arrowSize*0.5)+' '+(ax-dy*arrowSize*0.5)+','+(ay+dx*arrowSize*0.5));
        arrow.setAttribute('fill',lineColor);
        svg.appendChild(arrow);

        // Label pill
        var bg=document.createElementNS(ns,'rect');
        bg.setAttribute('x',mx-(labelWidth/2)); bg.setAttribute('y',my-11);
        bg.setAttribute('width',labelWidth); bg.setAttribute('height',22);
        bg.setAttribute('rx',11); bg.setAttribute('fill','#141820');
        bg.setAttribute('stroke',labelBorder); bg.setAttribute('stroke-width','1.2');
        svg.appendChild(bg);

        var t=document.createElementNS(ns,'text');
        t.setAttribute('x',mx); t.setAttribute('y',my+4.5);
        t.setAttribute('text-anchor','middle'); t.setAttribute('fill',labelColor);
        t.setAttribute('font-family','Figtree,sans-serif'); t.setAttribute('font-size','9');
        t.setAttribute('font-weight','700'); t.setAttribute('letter-spacing','0.8');
        t.textContent=labelText;
        svg.appendChild(t);
      });
    });
  }

  /* ── DELETE ── */
  window.lsmDeleteProperty=function(id){
    if(!confirm('Remove this property from the map?')) return;
    // Remove from group if any
    var group=lsmGetGroupOf(id);
    if(group){
      group.ids.delete(id);
      if(group.primaryId===id) group.primaryId=null;
      if(group.ids.size<2) lsmDissolveGroup(group);
    }
    lsmProps=lsmProps.filter(function(p){return p.id!==id;});
    var el=document.getElementById('lsm-prop-'+id);
    if(el){ el.style.transform='scale(0)'; el.style.opacity='0'; el.style.transition='all .2s'; setTimeout(function(){el.remove();},200); }
    lsmRerenderAll(); lsmUpdateLinks(); lsmUpdateSummary();
    if(lsmProps.length===0) document.getElementById('lsm-empty-state').style.display='';
    lsmShowHint('Property removed');
  };

  /* ── SUMMARY ── */

  function lsmFormatMoneyInput(input){
    input.addEventListener('input',function(){
      var pos=this.selectionStart;
      var raw=this.value.replace(/[^0-9.]/g,'');
      var parts=raw.split('.');
      var integer=parts[0].replace(/\B(?=(\d{3})+(?!\d))/g,',');
      var formatted=parts.length>1?integer+'.'+parts[1]:integer;
      var oldVal=this.value;
      var oldCommasBefore=(oldVal.slice(0,pos).match(/,/g)||[]).length;
      this.value=formatted;
      var newCommasBefore=(formatted.slice(0,pos).match(/,/g)||[]).length;
      var newPos=pos+(newCommasBefore-oldCommasBefore);
      try{this.setSelectionRange(newPos,newPos);}catch(e){}
    });
  }
